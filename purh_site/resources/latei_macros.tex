\NewDocumentCommand{\teiP}{O{} +m}{\par #2\par}

% Apply rend values predictably, including combined values in any order.
\NewDocumentCommand{\lateiHiBold}{m +m}{%
  \IfSubStr{#1}{bold}{\textbf{#2}}{#2}%
}
\NewDocumentCommand{\lateiHiSmallCaps}{m +m}{%
  \IfSubStr{#1}{small-caps}{\textsc{\lateiHiBold{#1}{#2}}}{\lateiHiBold{#1}{#2}}%
}
\NewDocumentCommand{\lateiHiInline}{m +m}{%
  \IfSubStr{#1}{italic}{\textit{\lateiHiSmallCaps{#1}{#2}}}{\lateiHiSmallCaps{#1}{#2}}%
}
\NewDocumentCommand{\teiHi}{O{} +m}{%
  \IfSubStr{#1}{sup}{\textsuperscript{\lateiHiInline{#1}{#2}}}{%
    \IfSubStr{#1}{sub}{$_{\lateiHiInline{#1}{#2}}$}{\lateiHiInline{#1}{#2}}%
  }%
}

% Nested footnotes are not valid LaTeX. If a teiNote appears inside another
% note, render a visible symbolic marker and inline parenthesized content.
\newif\iflateiinfootnote
\lateiinfootnotefalse
\NewDocumentCommand{\teiNote}{O{} +m}{%
  \iflateiinfootnote
    \textsuperscript{*}\,(#2)%
  \else
    \begingroup
      \lateiinfootnotetrue
      \footnote{#2}%
    \endgroup
  \fi
}

    }%
  }%
}
