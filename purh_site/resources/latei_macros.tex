% Experimental LaTEI macros for the reversible TEI/PURH convergence path.

% controlled LaTEI body produced by purh_site.reversible.latex_writer.

\usepackage{xparse}
\usepackage{xstring}

\NewDocumentCommand{\teiP}{O{} +m}{\par #2\par}
\NewDocumentCommand{\teiHead}{O{} +m}{\section*{#2}}
\NewDocumentCommand{\teiHi}{O{} +m}{%
  \IfSubStr{#1}{rend={italic}}{\textit{#2}}{%
    \IfSubStr{#1}{rend={small-caps}}{\textsc{#2}}{%
      \IfSubStr{#1}{rend={sup}}{\textsuperscript{#2}}{#2}%
    }%
  }%
}
\NewDocumentCommand{\teiNote}{O{} +m}{\footnote{#2}}
\NewDocumentCommand{\teiRef}{O{} +m}{#2}
\NewDocumentCommand{\teiTitle}{O{} +m}{%
  \IfSubStr{#1}{level={m}}{\textit{#2}}{%
    \IfSubStr{#1}{level={j}}{\textit{#2}}{%
      \IfSubStr{#1}{level={a}}{\enquote{#2}}{#2}%
    }%
  }%
}
